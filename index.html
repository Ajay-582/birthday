    function formatTime(ms) {
      const s = Math.floor(ms / 1000);
      const h = String(Math.floor(s / 3600)).padStart(2, '0');
      const m = String(Math.floor((s % 3600) / 60)).padStart(2, '0');
      const sec = String(s % 60).padStart(2, '0');

      function resize() {
        const rect = canvas.parentElement.getBoundingClientRect();
        canvas.width = W = rect.width;
        canvas.height = H = rect.height;
      }
      resize(); window.addEventListener('resize', resize);
      class Particle {
        constructor() { this.reset(); }
        reset() {
          this.x = rand(0, W); this.y = rand(0, H);
          this.size = rand(1.2, 3.6);
          this.speedX = rand(-0.25, 0.25);
          this.speedY = rand(-0.35, -0.08);
          this.opacity = rand(0.2, 0.7);
          this.hue = rand(330, 360);
        }
        update() {
          this.x += this.speedX; this.y += this.speedY;
          if (this.y < -20) this.reset();
          if (this.x < -20 || this.x > W + 20) this.reset();
        }
        draw() {
          ctx.beginPath();
          ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
          ctx.fillStyle = `hsla(${this.hue}, 80%, 70%, ${this.opacity * 0.5})`;
          ctx.fill();
          ctx.shadowColor = `hsla(${this.hue}, 80%, 70%, 0.2)`;
          ctx.shadowBlur = 12;
          ctx.fill();
          ctx.shadowBlur = 0;
        }
      }
      for (let i = 0; i < 80; i++) particles.push(new Particle());
      function animate() {
        ctx.clearRect(0, 0, W, H);
        particles.forEach(p => { p.update(); p.draw(); });
        requestAnimationFrame(animate);
      }
      animate();
    }

    function initFireflies() {
      const canvas = dom.firefliesCanvas;
      const ctx = canvas.getContext('2d');
      let W, H, fireflies = [];
      function resize() { canvas.width = W = window.innerWidth; canvas.height = H = window.innerHeight; }
      resize(); window.addEventListener('resize', resize);
      class Firefly {
        constructor() { this.reset(); }
        reset() {
          this.x = rand(0, W); this.y = rand(0, H);
          this.r = rand(2, 5);
          this.speed = rand(0.2, 0.6);
          this.angle = rand(0, Math.PI * 2);
          this.wobble = rand(0.02, 0.06);
          this.phase = rand(0, Math.PI * 2);
          this.alpha = rand(0.2, 0.8);
        }
        update() {
          this.angle += this.wobble;
          this.x += Math.cos(this.angle) * this.speed * 0.4;
          this.y += Math.sin(this.angle * 0.7) * this.speed * 0.3;
          this.phase += 0.03;
          if (this.x < -20 || this.x > W + 20 || this.y < -20 || this.y > H + 20) this.reset();
        }
        draw() {
          const glow = 0.4 + 0.6 * (0.5 + 0.5 * Math.sin(this.phase));
          const grad = ctx.createRadialGradient(this.x, this.y, 0, this.x, this.y, this.r * 5);
          grad.addColorStop(0, `rgba(255, 215, 0, ${glow * 0.6 * this.alpha})`);
          grad.addColorStop(1, `rgba(255, 215, 0, 0)`);
          ctx.fillStyle = grad;
          ctx.beginPath(); ctx.arc(this.x, this.y, this.r * 5, 0, Math.PI * 2); ctx.fill();
          ctx.beginPath(); ctx.arc(this.x, this.y, this.r * 0.6, 0, Math.PI * 2);
          ctx.fillStyle = `rgba(255, 235, 150, ${glow * 0.7 * this.alpha})`;
          ctx.fill();
        }
      }
      for (let i = 0; i < 35; i++) fireflies.push(new Firefly());
      function animate() {
        ctx.clearRect(0, 0, W, H);
        fireflies.forEach(f => { f.update(); f.draw(); });
        requestAnimationFrame(animate);
      }
      animate();
    }

    function initCounter() {
      const start = new Date(CONFIG.relationshipStart).getTime();
      const yearsEl = document.querySelector('[data-target-years]');
      const monthsEl = document.querySelector('[data-target-months]');
      const daysEl = document.querySelector('[data-target-days]');
      const liveEl = dom.liveTime;

      function update() {
        const now = Date.now();
        const diff = now - start;
        if (diff <= 0) return;

        const seconds = diff / 1000;
        const minutes = seconds / 60;
        const hours = minutes / 60;
        const days = hours / 24;
        const months = days / 30.44;
        const years = days / 365.25;
        yearsEl.textContent = Math.floor(years);
        monthsEl.textContent = Math.floor(months % 12);
        daysEl.textContent = Math.floor(days % 30.44);

        const startDate = new Date(start);
        const msSinceMidnight = now - new Date(new Date().getFullYear(), new Date().getMonth(), new Date().getDate()).getTime();
        const startTimeOfDay = startDate.getHours() * 3600000 + startDate.getMinutes() * 60000 + startDate.getSeconds() * 1000;
        const elapsedToday = (msSinceMidnight - startTimeOfDay + 86400000) % 86400000;
        liveEl.textContent = formatTime(elapsedToday);

        const totalSeconds = Math.floor(diff / 1000);
        const totalMinutes = Math.floor(totalSeconds / 60);
        const totalHours = Math.floor(totalMinutes / 60);
        const totalDays = Math.floor(totalHours / 24);
        const totalWeeks = Math.floor(totalDays / 7);
        const totalMonths = Math.floor(totalDays / 30.44);
        const totalYears = Math.floor(totalDays / 365.25);

        dom.spanWeeks.textContent = totalWeeks;
        dom.spanDays.textContent = totalDays;
        dom.spanHours.textContent = totalHours;
        dom.spanMinutes.textContent = totalMinutes;
        dom.spanSeconds.textContent = totalSeconds;
        dom.spanMonths.textContent = totalMonths;
        dom.spanYears.textContent = totalYears;

        const yearsRemain = totalYears;
        const monthsRemain = Math.floor(totalMonths % 12);
        const daysRemain = Math.floor(totalDays % 30.44);
        const hoursRemain = Math.floor(totalHours % 24);
        const minutesRemain = Math.floor(totalMinutes % 60);
        const secondsRemain = totalSeconds % 60;

        let parts = [];
        if (yearsRemain > 0) parts.push(`${yearsRemain} Year${yearsRemain>1?'s':''}`);
        if (monthsRemain > 0) parts.push(`${monthsRemain} Month${monthsRemain>1?'s':''}`);
        if (daysRemain > 0) parts.push(`${daysRemain} Day${daysRemain>1?'s':''}`);
        if (hoursRemain > 0) parts.push(`${hoursRemain} Hour${hoursRemain>1?'s':''}`);
        if (minutesRemain > 0) parts.push(`${minutesRemain} Minute${minutesRemain>1?'s':''}`);
        if (secondsRemain > 0 || parts.length === 0) parts.push(`${secondsRemain} Second${secondsRemain!==1?'s':''}`);

        dom.totalElapsed.textContent = parts.join(', ');

        requestAnimationFrame(update);
      }
      update();
    }

    function startFireworks() {
      const canvas = dom.fireworksCanvas;
      const ctx = canvas.getContext('2d');

      const colors = ['#ff4d6d','#ffd700','#ff8fab','#f3c98b','#ff6b8a','#ffaa00','#ff2d55','#ffcc02'];

      class FireworkParticle {
        constructor(x, y, color) {
          this.x = x; this.y = y; this.color = color;
          const angle = rand(0, Math.PI * 2);
          const speed = rand(2, 8);
          this.vx = Math.cos(angle) * speed;
          this.vy = Math.sin(angle) * speed;
          this.life = 1; this.decay = rand(0.006, 0.025);
          this.size = rand(2, 5); this.gravity = 0.06; this.trail = [];
        }
        update() {
          this.trail.push({ x: this.x, y: this.y });
          if (this.trail.length > 6) this.trail.shift();
          this.vx *= 0.99; this.vy += this.gravity;
          this.x += this.vx; this.y += this.vy;
          this.life -= this.decay; this.size *= 0.998;
        }
        draw() {
          this.trail.forEach((t, i) => {
            const alpha = (i / this.trail.length) * this.life * 0.3;
            ctx.beginPath(); ctx.arc(t.x, t.y, this.size * (i / this.trail.length) * 0.7, 0, Math.PI * 2);
            ctx.fillStyle = this.color.replace('1)', alpha + ')'); ctx.fill();
          });
          ctx.beginPath(); ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
          ctx.fillStyle = this.color.replace('1)', this.life + ')'); ctx.fill();
          const grad = ctx.createRadialGradient(this.x, this.y, 0, this.x, this.y, this.size * 4);
          grad.addColorStop(0, this.color.replace('1)', this.life * 0.4 + ')'));
          grad.addColorStop(1, this.color.replace('1)', '0)'));
          ctx.fillStyle = grad; ctx.beginPath(); ctx.arc(this.x, this.y, this.size * 4, 0, Math.PI * 2); ctx.fill();
        }
        get dead() { return this.life <= 0 || this.size < 0.3; }
      }

    // ---- Fireworks ----
    function startFireworks() {
      const canvas = dom.fireworksCanvas;
      const ctx = canvas.getContext('2d');
      let W, H;
      const particles = [];
      const colors = ['#ff4d6d','#ffd700','#ff8fab','#f3c98b','#ff6b8a','#ffaa00','#ff2d55','#ffcc02'];

      function resize() { canvas.width = W = window.innerWidth; canvas.height = H = window.innerHeight; }
      resize(); window.addEventListener('resize', resize);

      class FireworkParticle {
        constructor(x, y, color) {
          this.x = x; this.y = y; this.color = color;
          const angle = rand(0, Math.PI * 2);
          const speed = rand(2, 8);
          this.vx = Math.cos(angle) * speed;
          this.vy = Math.sin(angle) * speed;
          this.life = 1; this.decay = rand(0.006, 0.025);
          this.size = rand(2, 5); this.gravity = 0.06; this.trail = [];
        }
        update() {
          this.trail.push({ x: this.x, y: this.y });
          if (this.trail.length > 6) this.trail.shift();
          this.vx *= 0.99; this.vy += this.gravity;
          this.x += this.vx; this.y += this.vy;
          this.life -= this.decay; this.size *= 0.998;
        }
        draw() {
          this.trail.forEach((t, i) => {
            const alpha = (i / this.trail.length) * this.life * 0.3;
            ctx.beginPath(); ctx.arc(t.x, t.y, this.size * (i / this.trail.length) * 0.7, 0, Math.PI * 2);
            ctx.fillStyle = this.color.replace('1)', alpha + ')'); ctx.fill();
          });
          ctx.beginPath(); ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
          ctx.fillStyle = this.color.replace('1)', this.life + ')'); ctx.fill();
          const grad = ctx.createRadialGradient(this.x, this.y, 0, this.x, this.y, this.size * 4);
          grad.addColorStop(0, this.color.replace('1)', this.life * 0.4 + ')'));
          grad.addColorStop(1, this.color.replace('1)', '0)'));
          ctx.fillStyle = grad; ctx.beginPath(); ctx.arc(this.x, this.y, this.size * 4, 0, Math.PI * 2); ctx.fill();
        }
        get dead() { return this.life <= 0 || this.size < 0.3; }
      }

      function spawnBurst() {
        const x = rand(W * 0.1, W * 0.9);
        const y = rand(H * 0.1, H * 0.5);
        const color = colors[randInt(0, colors.length - 1)];
        const count = randInt(40, 90);
        for (let i = 0; i < count; i++) {
          const p = new FireworkParticle(x, y, color);
          p.size = rand(1.5, 4.5); p.decay = rand(0.005, 0.02);
          particles.push(p);
        }
      }

      function animate() {
        ctx.clearRect(0, 0, W, H);
        for (let i = particles.length - 1; i >= 0; i--) {
          const p = particles[i];
          p.update(); p.draw();
          if (p.dead) particles.splice(i, 1);
        }
        requestAnimationFrame(animate);
